```text
            if(pid < 0){

        exit(0);
    }

    /* wait for all child to terminate by checking the return value of wait */
    while((wait_return = wait(&status)) > 0){;}
```

```text
    return 0;
}
```

> **p.6** Combining fork() and exec()?
>
> Child process will have its own image, no longer a duplicate of its parent

**课件程序与接口（p.6，原页片段）**

```text
#include <sys/wait.h>
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>

int main(){

    pid_t pid;
    pid = fork();
    if(pid == -1){
        printf("Fork error\n");
        exit(-1);
    }
    else if(pid > 0){
        /* parent */
        printf("parent is doing something\n");
        sleep(5);
        printf("parent finished its task");
    }
    else{
        /* child */
        execlp("ls","ls","-a",NULL);
    }
    return 0;
}
```

p.4原例意图父进程循环创建10子进程，但截图实际先 `pid_t pid=0`，循环内又仅在 `pid>0`时fork，因此原样运行一次fork也不会发生，原进程最后反而走pid==0分支。要实现注释所说的10个子进程，必须先让父进程满足进入条件或重构分支；不能把注释当执行事实。假设意图修正且均成功，则新建10、连父共11，而不是每轮所有进程翻倍的1024。

p.5 exec名字里的l/v表示参数列表/数组，p表示按PATH寻找，e表示显式环境数组。例子envp仅给ABC=1；成功exec替换当前程序，后面的printf不执行，只有失败才返回。p.6先fork、子exec ls，父可继续自身逻辑；sleep(5)是延时，不是可靠等待子进程结束，应使用wait类接口管理结束。

**p.4 的原程序与旁边的文字目标不一致（已视觉核对）**：页上把 pid 初始化为 0，循环里却只在 `pid > 0` 时调用 fork。第一次测试是 `0 > 0`，结果为假；循环体未改变 pid，第二到第十次仍为假。因此按所展示代码执行，fork 一次也不会发生；循环结束后 `pid == 0` 成立，原进程自己打印 “Hello from child” 并 exit。它既不会生成注释声称的十个孩子，也不会走到最后的“All child terminated”。

要表达该页想教的“只有父进程继续创建”的逻辑，必须先让原父进程满足创建分支，并让成功 fork 后获得 0 的孩子退出创建路径。以父进程初始标记为正数的**教学修正**来推演：每轮父进程创建一个孩子，父分支得到正的孩子 PID，因而能进行下一轮；孩子得到 0，不再 fork；十轮后总计一个父进程和十个孩子。这里把课件原样与意图修正分开，未改动原课件，也不把有缺陷的原例当成已运行成功的程序。

**课件程序与接口（p.9，原页片段）**

```text
#include<stdio.h>
#include<stdlib.h>
#include<unistd.h>
#include<fcntl.h>

int main(){

    int pipe_fd[2];
    int bytes_read;
    int total_bytes_read = 0;
    char readbuf[1024];

    /* create a pipe */
    int ret = pipe(pipe_fd);
    if(ret < 0){
        printf("create pipe error\n");
        return 0;
    }

    /* write something into the pipe */
    write(pipe_fd[1], "------\npipe test1\n------\n", 25);
    write(pipe_fd[1], "pipe test2\n------\n", 18);

    /* close the write end, EOF will be added into the pipe */
    close(pipe_fd[1]);

    /* read from the pipe into a buffer until there is no content in the pipe */
    while ((bytes_read = read(pipe_fd[0], &readbuf[total_bytes_read], 1024-1-total_bytes_read)) > 0){
        total_bytes_read += bytes_read;
    }

    /* print the content in the buffer */
    printf("total read %d\n",total_bytes_read);
    readbuf[total_bytes_read]='\0';
    printf("%s\n", readbuf);

    /* close read end of pipe */
    close(pipe_fd[0]);

}
```

**课件程序与接口（p.9，原页片段）**

```text
Basic usage
Function prototype

Pipe() creates a pipe, a unidirectional data channel that can be used for inter-process communication.
pipefd[0]: read end of the pipe
pipefd[1]: write end of the pipe

```

dup返回新的可用fd，dup2把指定newfd指向同一open file description。原例 `dup2(fd,STDOUT_FILENO)`后printf仍写标准输出，只是目的地变文件。恢复前应先保存原stdout，例如 `saved=dup(1)`，重定向输出完成并适当fflush后 `dup2(saved,1)`再关闭saved；不是把原文件fd关掉就自动恢复终端。

图p.8先pipe再fork，父子都继承读写端；选择父写子读时，父关读端、子关写端。p.9先写25字节的第一串及18字节的第二串，再关写端，若两次write完整成功，共43字节；读循环累加实际返回值，末尾自行补NUL。EOF不是被写入管道的特殊字符，而是所有写端引用关闭且缓冲已读空时read返回0。

**回答 p.7 的原问题 “How to restore standard output after the dup2()?”**：重定向前用 `saved = dup(STDOUT_FILENO)` 保存旧标准输出的一份引用；确认成功后再 `dup2(fd, STDOUT_FILENO)`。需要恢复时先处理尚未输出的 stdio 缓冲，再 `dup2(saved, STDOUT_FILENO)`，确认恢复成功后 `close(saved)`。不能先关闭唯一的旧引用再试图从整数 1 找回它，因为 1 已经指向新文件。这里 saved 是另一个描述符，保存的是对旧打开对象的引用，不是把屏幕文字复制下来。

p.9 两次 write 的字符串按原字节计分别长 25 与 18，共 43B；都不包括字符串字面量末尾的零字节。read 可能分多次收齐，所以代码用 `total_bytes_read` 累加，并在数据末尾自行加 `\0`，才可用 `%s` 打印。原注释 “EOF will be added into the pipe” 容易误导：关闭最后一个写端没有塞入一个 EOF 字符；它改变的是内核端点引用状态。读完 43B 后、所有写端均关闭时，下一次 read 才返回 0。

> **p.11** Pipe w/ read/write
>
> example6.c
>
> If all file descriptors referring to the write end of a pipe have been closed
>
> Read from pipe will see end-of-file(EOF)
>
> If a process attempts to read from an empty pipe
>
> Read will block until data is available
>
> example7.c

**课件程序与接口（p.11，原页片段）**

```text
#include <assert.h>
#include <signal.h>
#include <stdio.h> /* perror */
#include <stdlib.h>
#include <sys/wait.h> /* wait, sleep */
#include <unistd.h> /* fork, write */

void signal_handler(int sig) {
    char s1[] = "SIGPIPE captured\n";
    char s2[] = "Unknown Signal\n";
    if (sig == SIGPIPE){
        write(STDOUT_FILENO, s1, sizeof(s1));
    }
    else{
        write(STDOUT_FILENO, s2, sizeof(s2));
    }
}

int main() {
    pid_t pid = getpid();
    int pipe_fd[2];

    /* create a pipe */
    int ret = pipe(pipe_fd);
    if(ret < 0){
        printf("create pipe error\n");
        return 0;
    }

    /* create a signal handler to capture signal pipe */
    signal(SIGPIPE, signal_handler);

    /* close all read end of the pipe */
    close(pipe_fd[0]);

    /* writer something to the pipe */
    write(pipe_fd[1], "test content", 12);

    return 0;
}
```

父写子读例先建立管道再fork，子关写端后读，父关读端后写 `Hello child process`的20字节（含末尾NUL），关闭写端，最后wait。子若忘关自己继承的写端，即使父已关闭，系统仍认为存在写者，读空后可能继续阻塞。

空管道且仍有写端时阻塞读；所有写端关闭且数据耗尽才EOF。所有读端关闭时写会引发SIGPIPE，若信号被捕获或忽略，write仍可能以EPIPE失败。满管道的阻塞写须等读者腾空间；“返回一次必读写全部请求”不是通用保证。

**补充：管道是否可读，要同时看缓冲区与所有写引用。** 缓冲区非空时，读者可以取得现有字节，即使所有写者已经退出；只有这些字节读完，正长度读取才会报告 EOF。缓冲区空但仍有写引用时，阻塞式读可能继续等待，因为系统不能排除将来还有字节到来。缓冲区空且所有写引用都关闭时，才具备 EOF 条件。

因此“最后一个写者关闭”不是把管道里尚未读出的内容一并删除，“没有新输出”也不是 EOF 的同义词。诊断卡住时用两个独立问题定位：管道还有多少待读数据；系统中是否仍有任何进程保留写引用。只在消费者那一行多加一次 read，无法修复父进程遗留写端造成的永久等待。

### 用描述符账本检查一条两段管道（补充）

设管道读端3、写端4，父进程fork两个孩子，分别作为生产者与消费者（为完整Shell结构自取的编号；不同于课件父进程本身exec的简化版本）。fork后每个进程都继承3和4。生产者dup2(4,1)后保留1这份写引用，关闭3与原4；消费者dup2(3,0)后保留0这份读引用，关闭原3与4；父管理者不传输数据，关闭自己3与4，再回收两个孩子。

| 进程 | 最后仍需的管道引用 | 为什么必须关其余引用 |
|---|---|---|
| 生产者 | stdout 1→写端 | 不该保留读端来掩盖“所有读者已退出” |
| 消费者 | stdin 0→读端 | 不该保留写端来阻止EOF |
| 父Shell | 无 | 不能因管理者遗留写端使消费者一直等 |

每次dup2后看引用，而不只看整数名：close(4)不会关闭生产者的1，因为它们是两个描述符引用。若dup2失败则不能继续exec假装已接好管道；若第二次fork失败，父侧也要关已建端点并处理第一个已启动孩子，不能只回到提示符留下后台残片。

:::hint 自测：消费者退出了，生产者为什么不一定立刻被系统终止？

只有生产者接下来向已无读端的管道写时才触发相关错误/信号条件；它若正做计算，未必立即观察到变化。若其他进程还遗留读端，甚至“没有读者引用”的条件还没成立。要查所有引用及实际调用，而不是只查预期消费者PID。

:::endhint
